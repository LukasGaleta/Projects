# Start from official Python image (small size, just enough to run Python 3.12)
FROM python:3.12-slim

# Set working directory inside container
WORKDIR /app

# Install Linux system dependencies (compiler, curl)
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential curl && \
    rm -rf /var/lib/apt/lists/*

# Copy requirements file first (better caching during build)
COPY requirements.txt .

# Install Python dependencies
RUN pip install --no-cache-dir -r requirements.txt

# Copy your app code into the container
COPY . .

# Tell fly.io your app listens on port 8080
EXPOSE 8080

# Command to start your Streamlit app
CMD ["streamlit", "run", "app.py", "--server.port=8080", "--server.address=0.0.0.0"]
